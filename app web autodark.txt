  <script>
    (function () {
      // Só rastreia se estiver no topo (evita iframes indesejados)
      if (window.self !== window.top) return;

      const APP_ID = "6a8b829cc4095a95e8bab08c";
      let lastPath = "";

      function getPageNameFromPath(path) {
        const segments = path.split("/").filter(Boolean);
        return segments[0] || "home";
      }

      function trackPageView() {
        const path = window.location.pathname;
        if (path === lastPath) return;
        lastPath = path;
        const pageName = getPageNameFromPath(path);

        // Envia log de forma assíncrona e não bloqueante
        fetch(`/api/app-logs/${APP_ID}/log-user-in-app/${pageName}`, {
          method: "POST",
          keepalive: true, // garante envio mesmo se a página fechar
        }).catch(() => {});
      }

      // Intercepta mudanças de rota em SPAs
      const originalPushState = history.pushState.bind(history);
      history.pushState = function (...args) {
        originalPushState(...args);
        trackPageView();
      };

      const originalReplaceState = history.replaceState.bind(history);
      history.replaceState = function (...args) {
        originalReplaceState(...args);
        trackPageView();
      };

      window.addEventListener("popstate", trackPageView);
      window.addEventListener("load", trackPageView);
      trackPageView(); // primeira visualização
    })();
  </script>
